cmake_minimum_required(VERSION 3.16)
project(BlochLab LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(BlochLab main.cpp)
if(APPLE)
  target_link_libraries(BlochLab PRIVATE "-framework OpenGL" "-framework GLUT")
elseif(WIN32)
  target_link_libraries(BlochLab PRIVATE opengl32 glu32 gdi32 user32)
  if(MSVC)
    set_property(TARGET BlochLab PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  elseif(MINGW)
    target_link_options(BlochLab PRIVATE -static -static-libgcc -static-libstdc++)
  endif()
else()
  message(FATAL_ERROR "Supported platforms: macOS and Windows")
endif()
enable_testing()
add_executable(quantum_tests tests.cpp)
add_test(NAME quantum_checks COMMAND quantum_tests)
